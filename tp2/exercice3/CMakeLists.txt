cmake_minimum_required(VERSION 3.10)
if(POLICY CMP0072)
  cmake_policy(SET CMP0072 NEW)
endif()
project(tp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32 AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Ce TP se compile avec gcc (CXX=${CMAKE_CXX_COMPILER_ID}).")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(TP_BIN tp)
else()
  file(RELATIVE_PATH TP_BIN "${CMAKE_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
  string(REPLACE "\\" "_" TP_BIN "${TP_BIN}")
  string(REPLACE "/" "_" TP_BIN "${TP_BIN}")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/tp_deps.cmake")
add_executable(${TP_BIN} src/main.cpp src/gl_utils.cpp src/scene.cpp)
set_target_properties(${TP_BIN} PROPERTIES OUTPUT_NAME tp RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(${TP_BIN} PRIVATE src)
tp_link_opengl(${TP_BIN})
tp_link_glfw(${TP_BIN})
tp_link_glm(${TP_BIN})
target_compile_definitions(${TP_BIN} PRIVATE SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
if(MSVC)
  target_compile_options(${TP_BIN} PRIVATE /W4)
else()
  target_compile_options(${TP_BIN} PRIVATE -Wall -Wextra -Wpedantic)
endif()
